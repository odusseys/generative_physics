The projection equation is:

```python

```

For PCG, solve the positive semidefinite system:

```python
def _masked_positive_laplacian(phi, geometry):
    dx = geometry["dx"]
    dy = geometry["dy"]
    dx2 = dx * dx
    dy2 = dy * dy

    open_e = geometry["open_e"].to(dtype=phi.dtype)
    open_w = geometry["open_w"].to(dtype=phi.dtype)
    open_s = geometry["open_s"].to(dtype=phi.dtype)
    open_n = geometry["open_n"].to(dtype=phi.dtype)
    denom = geometry["denom"].to(dtype=phi.dtype)
    fluid = geometry["fluid_float"].to(dtype=phi.dtype)

    neighbor_sum = torch.zeros_like(phi)

    neighbor_sum[:, :-1] += open_e[:, :-1] * phi[:, 1:] / dx2
    neighbor_sum[:, 1:] += open_w[:, 1:] * phi[:, :-1] / dx2
    neighbor_sum[:-1, :] += open_s[:-1, :] * phi[1:, :] / dy2
    neighbor_sum[1:, :] += open_n[1:, :] * phi[:-1, :] / dy2

    out = denom * phi - neighbor_sum
    out = out * fluid
    out = _zero_mean_fluid(out, geometry)
    return out
```

This operator should correspond to homogeneous Neumann on solid boundaries and no pressure coupling through obstacle faces.

Implementation requirements:

* Convert `rhs` to zero-mean over fluid cells.
* Solve `A phi = b`, where `b = -rhs`.
* Use `_masked_positive_laplacian`.
* Use Jacobi preconditioning with `inv_diag`.
* Warm-start from `initial_phi` when provided.
* Stop when:

```python

```

* Always return a zero-mean fluid-only `phi`.

Skeleton:

```python
def _masked_poisson_pcg(rhs, geometry, max_iterations=80, tolerance=1e-4, initial_phi=None):
    fluid = geometry["fluid_float"].to(dtype=rhs.dtype)
    inv_diag = geometry["inv_diag"].to(dtype=rhs.dtype)

    rhs = _zero_mean_fluid(rhs, geometry)
    b = _zero_mean_fluid(-rhs, geometry)

    if initial_phi is None:
        x = torch.zeros_like(rhs)
    else:
        x = _zero_mean_fluid(initial_phi.to(device=rhs.device, dtype=rhs.dtype), geometry)

    r = _zero_mean_fluid(b - _masked_positive_laplacian(x, geometry), geometry)
    z = _zero_mean_fluid(inv_diag * r, geometry)
    p = z.clone()

    rz_old = (r * z).sum()
    b_norm = torch.sqrt((b * b).sum()).clamp_min(1e-20)

    if torch.sqrt((r * r).sum()) <= float(tolerance) * b_norm:
        return x

    for _ in range(int(max_iterations)):
        Ap = _masked_positive_laplacian(p, geometry)
        denom = (p * Ap).sum().clamp_min(1e-20)
        alpha = rz_old / denom

        x = _zero_mean_fluid(x + alpha * p, geometry)
        r = _zero_mean_fluid(r - alpha * Ap, geometry)

        if torch.sqrt((r * r).sum()) <= float(tolerance) * b_norm:
            break

        z = _zero_mean_fluid(inv_diag * r, geometry)
        rz_new = (r * z).sum()
        beta = rz_new / rz_old.clamp_min(1e-20)
        p = _zero_mean_fluid(z + beta * p, geometry)
        rz_old = rz_new

    return _zero_mean_fluid(x, geometry)
```
